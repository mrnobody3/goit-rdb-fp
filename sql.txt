CREATE DATABASE pandemic;

USE pandemic;

CREATE TABLE entities (
    entity_id INT AUTO_INCREMENT PRIMARY KEY,
    entity_name VARCHAR(150) NOT NULL UNIQUE,
    code VARCHAR(10) DEFAULT NULL
);


INSERT INTO entities (entity_name, code)
SELECT DISTINCT Entity, Code
FROM infectious_cases;


CREATE TABLE normalized_infectious_cases (
    case_id INT AUTO_INCREMENT PRIMARY KEY,
    entity_id INT NOT NULL,
    year_case INT NOT NULL,
    number_yaws DOUBLE DEFAULT NULL,
    polio_cases DOUBLE DEFAULT NULL,
    cases_guinea_worm DOUBLE DEFAULT NULL,
    number_rabies DOUBLE DEFAULT NULL,
    number_malaria DOUBLE DEFAULT NULL,
    number_hiv DOUBLE DEFAULT NULL,
    number_tuberculosis DOUBLE DEFAULT NULL,
    number_smallpox DOUBLE DEFAULT NULL,
    number_cholera_cases DOUBLE DEFAULT NULL,


    CONSTRAINT fk_cases_entity
        FOREIGN KEY (entity_id) REFERENCES entities(entity_id)
        ON DELETE CASCADE ON UPDATE CASCADE,


    UNIQUE KEY uq_entity_year (entity_id, year_case)
);


INSERT INTO normalized_infectious_cases (
    entity_id,
    year_case,
    number_yaws,
    polio_cases,
    cases_guinea_worm,
    number_rabies,
    number_malaria,
    number_hiv,
    number_tuberculosis,
    number_smallpox,
    number_cholera_cases
)
SELECT
    e.entity_id,
    r.Year,
    NULLIF(r.Number_yaws, ''),
    NULLIF(r.polio_cases, ''),
    NULLIF(r.cases_guinea_worm, ''),
    NULLIF(r.Number_rabies, ''),
    NULLIF(r.Number_malaria, ''),
    NULLIF(r.Number_hiv, ''),
    NULLIF(r.Number_tuberculosis, ''),
    NULLIF(r.Number_smallpox, ''),
    NULLIF(r.Number_cholera_cases, '')
FROM infectious_cases r
JOIN entities e ON r.Entity = e.entity_name;


SELECT COUNT(*) FROM normalized_infectious_cases;


-- 3
SELECT
    e.entity_id,
    e.entity_name,
    e.code,
    AVG(c.number_rabies) AS avg_rabies,
    MIN(c.number_rabies) AS min_rabies,
    MAX(c.number_rabies) AS max_rabies,
    SUM(c.number_rabies) AS sum_rabies
FROM normalized_infectious_cases c
JOIN entities e ON c.entity_id = e.entity_id
WHERE c.number_rabies IS NOT NULL
GROUP BY e.entity_id, e.entity_name, e.code
ORDER BY avg_rabies DESC
LIMIT 10;

-- 4
SELECT
    Entity,
    Code,
    AVG(CAST(Number_rabies AS DOUBLE)) AS avg_rabies,
    MIN(CAST(Number_rabies AS DOUBLE)) AS min_rabies,
    MAX(CAST(Number_rabies AS DOUBLE)) AS max_rabies,
    SUM(CAST(Number_rabies AS DOUBLE)) AS sum_rabies
FROM infectious_cases
WHERE Number_rabies IS NOT NULL
  AND Number_rabies != ''
GROUP BY Entity, Code
ORDER BY avg_rabies DESC
LIMIT 10;


-- 5
DROP FUNCTION IF EXISTS calculate_year_diff;

DELIMITER //

CREATE FUNCTION calculate_year_diff(input_year INT)
RETURNS INT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE date_from_year DATE;
    DECLARE years_difference INT;

    -- Створюємо дату 1 січня переданого року
    SET date_from_year = MAKEDATE(input_year, 1);

    -- Розраховуємо різницю в роках відносно поточної дати
    SET years_difference = TIMESTAMPDIFF(YEAR, date_from_year, CURDATE());

    RETURN years_difference;
END //

DELIMITER ;

SELECT
    year_case,
    MAKEDATE(year_case, 1) AS start_of_year,
    calculate_year_diff(year_case) AS diff_in_years
FROM (
    SELECT DISTINCT year_case
    FROM normalized_infectious_cases
) AS unique_years
ORDER BY year_case DESC;
